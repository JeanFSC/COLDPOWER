function takeScreenshot(urlPath: string, fileName: string, viewport = "1280,720") {
  const fullUrl = `http://localhost:3007${urlPath.startsWith("/") ? urlPath : "/" + urlPath}`;
  const targetPath = path.join(CAPTURES_DIR, fileName);
  try {
    console.log(`[CAPTURE] ${fileName} from ${urlPath}...`);
    execSync(`npx playwright screenshot --viewport-size="${viewport}" --wait-for-timeout=2000 "${fullUrl}" "${targetPath}"`, {
      stdio: "pipe",
      timeout: 30000,
    });
  } catch (error) {
    console.warn(`[CAPTURE WARNING] Failed ${fileName}:`, error instanceof Error ? error.message : error);
  }
}

export interface AuditStepResult {
  step: string;
  actor: "CLIENTE" | "VENTAS" | "ALMACEN" | "CAJA" | "GERENCIA" | "COMPRAS" | "CATALOGO" | "SISTEMA";
  action: string;
  expected: string;
  obtained: string;
  passed: boolean;
  sqlVerifications: Array<{ query: string; result: unknown }>;
  severity?: "P0" | "P1" | "P2" | "P3";
  defect?: string;
  screenshot?: string;
}

export interface JourneyAuditResult {
  code: "C1" | "C2" | "C3" | "C4" | "C5" | "C6" | "C7" | "C8" | "C9" | "C10";
  title: string;
  verdict: "PASSED" | "FAILED" | "DEFECTS_FOUND";
  steps: AuditStepResult[];
  defects: string[];
}
